Статическим методы:

Метод можно объявить статическим безболезненно только в том случае, если он удовлетворяет всем следующим условиям:
1. Чистая функция (Pure Function): Метод зависит только от входящих аргументов, не читает и не меняет внешнее состояние (базу данных, конфигурацию, файлы).
2. Отсутствие бизнес-логики: Это утилитарная операция (например, конвертация строки, валидация формата email, математическое вычисление).
3. Неизменность алгоритма: Поведение метода гарантированно не изменится в зависимости от контекста выполнения или будущего расширения системы.

Как замокать метод в writeToFileSystem() в интеграционном тесте:

package ru.ruslan.service.impl;

import lombok.extern.slf4j.Slf4j;
import org.springframework.core.io.ClassPathResource;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.InvalidPathException;
import java.nio.file.Path;
import java.nio.file.Paths;

@Slf4j
@Service
public class FileService {

    public void writeToFileSystem(String filePath, String content) {
        try {
            Path path = Path.of(filePath);
            if (path.getParent() != null) {
                Files.createDirectories(path.getParent());
            }
            Files.writeString(path, content, StandardCharsets.UTF_8);
            log.info("Файл успешно создан/обновлен по адресу: {}", filePath);
        } catch (IOException e) {
            log.error("Ошибка записи файла в ФС по пути {}: {}", filePath, e.getMessage());
            throw new RuntimeException("Не удалось записать файл в ФС: " + filePath, e);
        }
    }

}
